  <script>
    const WHATSAPP = "https://w.app/confirmacaochadepanela";
    const TARGET = new Date("2026-10-10T14:00:00-03:00").getTime();
    const RESERVAS_KEY = "cha-de-panela-reservas";

    // Contagem regressiva
    function updateTimer() {
      const total = Math.max(0, TARGET - Date.now());
      const dias = Math.floor(total / 86400000);
      const horas = Math.floor((total / 3600000) % 24);
      const min = Math.floor((total / 60000) % 60);
      const seg = Math.floor((total / 1000) % 60);
      document.getElementById("dias").textContent = String(dias).padStart(2, "0");
      document.getElementById("horas").textContent = String(horas).padStart(2, "0");
      document.getElementById("min").textContent = String(min).padStart(2, "0");
      document.getElementById("seg").textContent = String(seg).padStart(2, "0");
    }
    updateTimer();
    setInterval(updateTimer, 1000);

    // Lista de presentes
    const slug = s => s.normalize("NFD").replace(/[\u0300-\u036f]/g, "").toLowerCase().replace(/[^a-z0-9]+/g, "-").replace(/^-|-$/g, "");
    const g = (title, names) => ({ title, items: names.map(name => ({ id: `${slug(title)}-${slug(name)}`, name })) });
    const groups = [
      g("Cozinha", ["Abridor de latas","Açucareiro","Assadeiras","Descanso de panela","Escorredor de louça","Faqueiro","Forma para bolos","Fruteira","Jogo de copos","Jogo de panelas","Jogo de pratos (rasos e fundos)","Jogo de sobremesa","Jogo de talheres","Jogo de xícaras","Pano de prato","Peneira","Porta detergente","Porta mantimentos (potes)","Potes de plástico","Potes de vidro","Ralador","Rodo de pia","Saleiro grande","Tábua de corte","Abridor de garrafa","Amassador de batatas","Boleira","Travessas de vidro quadrada","Travessa de vidro redonda","Cesto de pão","Chaleira elétrica","Cumbucas","Descascador","Escorredor de macarrão","Espremedor de limão/alho","Forma de gelo de silicone","Funil","Garrafa de água de vidro","Garrafa térmica","Jarra de suco de vidro","Jogo americano","Potes para café/açúcar","Lixeira de cozinha","Panela de pressão 3L","Frigideira de ferro","Panela de ferro","Pipoqueira","Porta frios","Porta guardanapo","Porta xícaras","Suporte guardanapo","Suporte papel toalha","Tábuas de madeira / petisqueira","Tapetes de cozinha","Triturador de alimentos manual","Utensílios de silicone","Liquidificador","Sanduicheira","Batedeira","Air fryer","Mixer"]),
      g("Banheiro", ["Toalhas de banho","Toalhas de rosto","Jogo de toalhas","Cortina para box","Tapetes","Saboneteira","Porta escovas de dente","Porta algodão","Porta cotonetes","Lixeira","Cesto de roupas sujas","Dispenser para sabonete líquido"]),
      g("Lavanderia", ["Varal (de chão ou de teto)","Cesto de roupas sujas","Baldes","Bacia plástica","Esfregão / mop","Cabides","Tábua de passar roupa","Ferro de passar"]),
      g("Quarto", ["Kit roupa de cama","Almofadas decorativas","Travesseiros","Jogo de cama","Edredom"]),
    ];
    const totalItems = groups.reduce((n, g) => n + g.items.length, 0);

    let reserved = {};
    try {
      const raw = localStorage.getItem(RESERVAS_KEY);
      if (raw) reserved = JSON.parse(raw);
    } catch (e) { reserved = {}; }

    let openId = null;

    function saveReserved(next) {
      reserved = next;
      localStorage.setItem(RESERVAS_KEY, JSON.stringify(reserved));
      renderGifts();
    }

    function renderGifts() {
      const taken = Object.keys(reserved).length;
      document.getElementById("gift-status").textContent = `${taken} de ${totalItems} itens reservados`;
      const container = document.getElementById("gift-list");
      container.innerHTML = "";

      groups.forEach(group => {
        const div = document.createElement("div");
        div.className = "gift-group";
        div.innerHTML = `<h3>${group.title}</h3><ul></ul>`;
        const ul = div.querySelector("ul");

        group.items.forEach(gift => {
          const by = reserved[gift.id];
          const isOpen = openId === gift.id;
          const li = document.createElement("li");

          const checkbox = document.createElement("div");
          checkbox.className = "checkbox" + (by ? " checked disabled" : "");
          checkbox.setAttribute("role", "checkbox");
          checkbox.setAttribute("aria-checked", by ? "true" : "false");
          checkbox.setAttribute("aria-label", `Reservar ${gift.name}`);
          checkbox.innerHTML = by ? "✓" : "";
          checkbox.onclick = () => {
            if (by) return;
            openId = isOpen ? null : gift.id;
            renderGifts();
          };
          li.appendChild(checkbox);

          const info = document.createElement("div");
          info.className = "gift-name" + (by ? " reserved" : "");
          info.innerHTML = `<p>${gift.name}</p>` + (by ? `<p class="gift-reserved-by">Reservado por ${by}</p>` : "");
          li.appendChild(info);

          if (!by && isOpen) {
            const form = document.createElement("form");
            form.className = "reserve-form";
            form.innerHTML = `
              <input type="text" placeholder="Seu nome" required autofocus />
              <button type="submit" class="btn btn-primary">Reservar</button>
            `;
            form.onsubmit = (e) => {
              e.preventDefault();
              const name = form.querySelector("input").value.trim();
              if (!name) return;
              saveReserved({ ...reserved, [gift.id]: name });
              openId = null;
            };
            li.appendChild(form);
          }

          ul.appendChild(li);
        });

        container.appendChild(div);
      });
    }
    renderGifts();

    // RSVP
    const rsvpForm = document.getElementById("rsvp-form");
    const rsvpCard = document.getElementById("rsvp-card");
    rsvpForm.onsubmit = (e) => {
      e.preventDefault();
      const nome = document.getElementById("rsvp-nome").value.trim();
      const pessoas = document.getElementById("rsvp-pessoas").value;
      if (!nome) return;
      const primeiro = nome.split(" ")[0];
      window.open(WHATSAPP, "_blank", "noopener,noreferrer");
      rsvpCard.innerHTML = `
        <div class="success-card">
          <h3>Presença confirmada, ${primeiro}!</h3>
          <p>Anotamos ${pessoas} ${Number(pessoas) > 1 ? "pessoas" : "pessoa"}. Abrimos o WhatsApp pra você enviar a confirmação — se não abrir, use o botão abaixo.</p>
          <a href="${WHATSAPP}" target="_blank" rel="noreferrer" class="btn btn-primary" style="margin-top: 24px;">Enviar confirmação no WhatsApp</a>
        </div>
      `;
    };
  </script>
